def max_ltv(app):
    if is_grandfathered(app, application_date <= GRANDFATHERING_CUTOFF):
        return 0.70          # 종전규정 유지
    if region_status(app) != REGULATED:

    if app.house_count >= 2:       return 0.00   # 다주택
